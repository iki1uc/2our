export function ID_2our(vec) { return { mode: "vector-id", identity: "Vektor-Operator", magnitude: Math.sqrt(vec.x**2 + vec.y**2 + vec.z**2), direction: { x: vec.x / Math.sqrt(vec.x**2 + vec.y**2 + vec.z**2), y: vec.y / Math.sqrt(vec.x**2 + vec.y**2 + vec.z**2), z: vec.z / Math.sqrt(vec.x**2 + vec.y**2 + vec.z**2) } }; }
